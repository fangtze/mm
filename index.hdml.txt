function ri(a,b){return Math.floor(Math.random()*(b-a+1))+a}
function shuffle(a){return a.sort(()=>Math.random()-.5)}
function choices(ans,min,max){
 let a=[String(ans)];while(a.length<4){let n=String(ri(min,max));if(!a.includes(n))a.push(n)}return shuffle(a)
}
function qmake(){
 let type=topic;if(type==="random")type=topics.filter(x=>x.id!=="random")[ri(0,11)].id;
 let q,a,ch;
 if(type==="composition"){let total=ri(6,20),x=ri(1,total-1);q=t(`${total} 可以分成 ${x} 和 ［？］`,`${total} can be split into ${x} and [?]`);a=total-x;ch=choices(a,0,total)}
 if(type==="pattern"){let start=ri(1,10),step=ri(1,5);q=t(`找规律：${start}，${start+step}，${start+2*step}，［？］`,`Find the pattern: ${start}, ${start+step}, ${start+2*step}, [?]`);a=start+3*step;ch=choices(a,1,30)}
 if(type==="approx"){let n=ri(11,99),r=n%10>=5?Math.ceil(n/10)*10:Math.floor(n/10)*10;q=t(`${n} 约等于多少？（最接近的十位数）`,`What is ${n} approximately? (Nearest ten)`);a=r;ch=choices(a,0,100)}
 if(type==="addition100"){let x=ri(1,60),y=ri(1,100-x);q=t(`${x} + ${y} = ？`,`${x} + ${y} = ?`);a=x+y;ch=choices(a,0,100)}
 if(type==="subtraction100"){let x=ri(10,100),y=ri(1,x);q=t(`${x} − ${y} = ？`,`${x} − ${y} = ?`);a=x-y;ch=choices(a,0,100)}
